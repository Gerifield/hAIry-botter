func biggestImage(photos []models.PhotoSize) models.PhotoSize {
	if len(photos) == 0 {
		return models.PhotoSize{}
	}

	biggest := photos[0]
	for _, photo := range photos {
		if photo.FileSize > biggest.FileSize {
			biggest = photo
		}
	}

	return biggest
}
